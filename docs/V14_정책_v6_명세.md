## 3. commissionCalc.js — 새 calc_method

### 새 함수 추가
```js
// 예상금액비율 (쿨가이 가스 KA/KB)
export function calcEstimateSplit(opts) {
  const { policy, estimate, extra = 0 } = opts;
  const principalRate = policy.principal.value / 100;  // KA 0.10 / KB 0.35
  const engineerRate  = policy.engineer.value / 100;   // 0.50

  const principal = Math.round(estimate * principalRate);
  const engineer  = Math.round(estimate * engineerRate);
  const company   = estimate - principal - engineer;

  // 현장 추가금: 사장님 50% + 기사 50% (원청 0%)
  const extraEng = Math.round(extra * 0.5);
  const extraOwn = extra - extraEng;

  return {
    total: estimate + extra,
    engineer: engineer + extraEng,
    principal,
    company:  company + extraOwn,
  };
}

// 비율_총액기준 (쿨가이 가스 1way/4way/원형 2+)
export function calcRatioTotalBase(opts) {
  const { policy, estimate, qty, extra = 0 } = opts;
  const totalBase = estimate * qty + extra;
  const principal = Math.round(totalBase * (policy.principal / 100));
  const engineer  = Math.round(totalBase * (policy.engineer / 100));
  const company   = totalBase - principal - engineer;
  return { total: totalBase, engineer, principal, company };
}

// AG전체_AD반반 (유솔N 냉매점검)
export function calcAgFullAdHalf(opts) {
  const { unitPrice = 10000, qty = 1, extra = 0, travel = 0 } = opts;
  const baseTotal = unitPrice * qty;
  let engineer  = 0;
  let principal = baseTotal;  // 기본 100%
  let company   = 0;

  // 추가금: 사장님 50% + 기사 50%
  const extraEng = Math.round(extra * 0.5);
  engineer += extraEng;
  company  += extra - extraEng;

  // 출장: 기사 100%
  engineer += travel;

  return {
    total: baseTotal + extra + travel,
    engineer, principal, company,
  };
}

// 추가선택 (유솔N 송풍팬/실외기/피톤치드)
export function calcAdditional(opts) {
  const { unitPrice, qty = 1 } = opts;
  const total     = unitPrice * qty;
  const engineer  = Math.round(total * 0.85);
  const principal = total - engineer;
  return { total, engineer, principal, company: 0 };
}
```

  const code = getServiceCode(task);
  const principal = findPrincipalByClient(task.client, principalsList);

  if (code === "visit") return { ...visitFee };

  if (principal.id === "usol_n") {

      policy,
      estimate: task.estimateTotal,
      extra: task.addonFee || 0,
    });
