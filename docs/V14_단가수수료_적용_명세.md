## 4. EngineerTaskCompletionScreens.jsx — `FEE_RATE` 제거

### 옛
```js
const FEE_RATE = 0.30;
const rate = FEE_RATE;
const fee = Math.floor(estimateTotal × rate);
const earning = estimateTotal - fee;
```

### 새
```js
import { calcCleaning, calcRefrigerant } from "../utils/commissionCalc.js";
import { loadPrincipals } from "../data/principals.js";

function calcTaskEarning(task) {
  const principals = loadPrincipals();
  const principal = principals.find(p => p.name === task.client || p.nickname === task.client);
  if (!principal) return { engineer: task.engineerNet || 0, principal: 0, company: 0 };
  
  const isRefrigerant = (task.workType || "").includes("냉매") || (task.workType || "").includes("충전");
  const policy = isRefrigerant ? principal.commissionPolicy.refrigerant : principal.commissionPolicy.cleaning;
  const total = (task.estimateTotal || 0) + (task.addonFee || 0) + (task.extraFee || 0);
  
  if (isRefrigerant) {
    return calcRefrigerant({ policy, estimate: task.estimateTotal || 0, extra: (task.addonFee || 0) + (task.extraFee || 0), engineerId: task.assignedEngineerId });
  }
  const appliances = [{ type: task.appliance, count: task.qty || 1 }];
  return calcCleaning({ policy, principal, appliances, total, additionals: task.additionals || [] });
}
```

→ 화면 표시: 기사에게는 `result.engineer` 만 노출. `principal` / `company` X (쿨가이 비밀 유지).

### 새 (정확)
```js
{
  date: "2026-05-04", status: "pending",
  works: [
    { id: "A260504-001", customerName: "박지영", workType: "세척", workItem: "벽걸이", quantity: 1,
      feeAmount: 40000 }   // 직영 80,000 - 기사 40,000 = 회사 송금 40,000
  ],
  totalAmount: 40000,    // works 합계 자동

}
```

**계산 정의**:
- `feeAmount` (회사 송금) = `total - engineer` (= principal + company)
- 일별 `totalAmount` = 그날 작업 `feeAmount` 합계

### 유솔N의 다른 점
- 유솔N은 회사 송금 X, 다음 달 15일 직접 입금
- `usolNGroupsMock`의 `feeAmount` = 기사가 받는 돈 (= calcCleaning result.engineer)
- `totalAmount` = 그날 기사 받을 돈 합계
